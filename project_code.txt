function getTableField(table, id, field) {
    let tableName = table + "_table";

    if (tables[tableName]) {

        let item = tables[tableName].find(x => x[table + "_ID"] == id)
        //console.log(item)
        if (item && item[field]) {
            return item[field]
        }
    }
    return ""
}

function getTableFieldUtr(table, id, field, idName) {
    let tableName = table + "_table";

    if (tables[tableName]) {

        let item = tables[tableName].find(x => x[idName] == id)
        //console.log(item)
        if (item && item[field]) {
            return item[field]
        }
    }
    return ""
}

function findItem(id) {
    let item = tables.Bridge_table.find(x => x.Bridge_DayId == id);
    selectedItem.value = item;

    return item;
}

function getProp(name, val) {
    console.log(`${name} - ${val}`);
    
    let tbl = tables[name + "_table"]
    if (tbl) {
        let itemID = name + "_ID"
        if (tbl[0][itemID]) {

            let item = tables[name + "_table"].find(x => x[itemID] == val);
            return JSON.stringify(item);
        }
        else {
            return "no id"
        }

    }
    else {
        return 'not a table'
    }
}

function getIdProp(obj) {
    var props = Object.getOwnPropertyNames(obj);
    var id = props.find(x => x.indexOf("ID") != -1);

    return id;
}

function refactorDB() {
    const readingFixArray = tables.Bridge_table.map(item => ({

        .filter(item => item.Day.includes('звычай')) // Фильтруем по ключевой фразе
        .reduce((accumulator, current) => {
            const isDuplicate = accumulator.find(item => item.Day === current.Day && item.weekDay === current.weekDay);
            if (!isDuplicate) {
                accumulator.push({ date: current.date, Day: current.Day, weekDay: current.weekDay, officeOfReadingsVers: current.officeOfReadingsVers });
            }
            return accumulator;
        }, []);

    console.log(`unique elems ${ordinaryVersArray.length}`);

    const daysCollectionRef = collection(db, 'days');

    // Получение документов из Firestore и обновление их по данным из uniqueDays
    async function updateFirestoreDays(uniqueDays) {
        try {
            const querySnapshot = await getDocs(query(daysCollectionRef, orderBy('_weekNum')));
            const batch = [];

            // Проходим по каждому документу из Firestore
            querySnapshot.forEach(docSnapshot => {
                const docData = docSnapshot.data();
                const matchingUniqueDay = uniqueDays.find(uniqueDay => uniqueDay.Day === docData._dayName && uniqueDay.weekDay === docData._weekDay);

                // Если найден соответствующий uniqueDay, обновляем officeOfReadingsVers
                if (matchingUniqueDay) {
                    const docRef = doc(db, 'days', docSnapshot.id);
                    const updateOperation = updateDoc(docRef, {
                        officeOfReadingsVers: matchingUniqueDay.officeOfReadingsVers
                    });
                    // console.log(`${docData._dayName}-${docData._weekDay} updated`)
                    batch.push(updateOperation);
                }
            });

            console.log(`to update ${batch.length}`);
            // Выполняем все обновления
            await Promise.all(batch);

            console.log("Обновление завершено.");
        } catch (error) {
            console.error("Ошибка при обновлении Firestore:", error);
        }
    }

    // Вызов функции
    updateFirestoreDays(ordinaryVersArray);
}

            const ash = getMomentDay(opts.ashWednesday, opts.year);
            const easter = getMomentDay(opts.easterSunday, opts.year);

            if (current.diff(ash, "days") >= 0 && current.diff(easter, "days") < 0) {
                const firstLentSunday = ash.add(4, "days");
                const lastLentSunday = easter.add(-1, "weeks");

                const startDiff = current.diff(firstLentSunday, "days");
                const endDiff = current.diff(lastLentSunday, "days");
                if (startDiff < 0) {

                    logDay('after ash')
                    await getItemsForPeriod(period, 7 + startDiff + 1, 0)

                } else if (endDiff > 0 && endDiff < 8) {
                    // great week
                    let holyWeekDiff = current.diff(lastLentSunday, "days")
                    let holyWeekPeriod = 'holyweek'

                    await getItemsForPeriod(holyWeekPeriod, endDiff + 1, 1);

                } else if (endDiff == 0) {

                    await getDayByFeast("palmSunday");
                    // palm sunday
                } else {
                    logDay('lent time')
                    // 1 - sunday
                    // 7 - saturday
                    let dayOfWeek = startDiff % 7 + 1
                    let periodWeek = Math.floor(startDiff / 7) + 1
                    await getItemsForPeriod(period, dayOfWeek, periodWeek);
                }
            }
        }

        async function checkEaster(opts, current) {
            const period = "easter"
            const easter = getMomentDay(opts.easterSunday, opts.year);
            //const ascension = getMomentDay(opts.ascension, opts.year);
            const pentecost = getMomentDay(opts.pentecost, opts.year);

            if (current.diff(easter, "days") >= 0 && current.diff(pentecost, "days") < 0) {
                logDay("easter time")

                const diff = current.diff(easter, "days");
                // 1 - sunday
                // 7 - saturday
                const dayOfWeek = diff % 7 + 1
                const periodWeek = Math.floor(diff / 7) + 1

                await getItemsForPeriod(period, dayOfWeek, periodWeek);
            }
        }

        async function checkOrdinary(opts, current) {
            const period = "ordinary"
            const ash = getMomentDay(opts.ashWednesday, opts.year);
            //const ascension = getMomentDay(opts.ascension, opts.year);
            const pentecost = getMomentDay(opts.pentecost, opts.year);
            const fa = getMomentDay(opts.firstAdventSunday, opts.year);
            const weeksBefore = parseInt(opts.ordintaryWeeks1) - 1;
            const fo = ash.clone().add(-2, "days").add(-weeksBefore, "weeks");

            if (current.diff(fo, "days") >= 0 && current.diff(ash, "days") < 0) {
                logDay("ordinary before ash")
                let diff = current.diff(fo, "days");
                // 1 - sunday
                // 7 - saturday
                let dayOfWeek = diff % 7 + 1
                let periodWeek = Math.floor(diff / 7) + 1

                await getItemsForPeriod(period, dayOfWeek, periodWeek);
            } else if (current.diff(pentecost, "days") > 0 && current.diff(fa, "days") < 0) {
                let diff = current.diff(pentecost, "days");
                // 1 - sunday
                // 7 - saturday
                let startWeek = parseInt(opts.ordintaryWeeks2);

                let dayOfWeek = diff % 7 + 1
                let periodWeek = Math.floor(diff / 7) + startWeek

                logDay(`ordinary after pentocost. week - ${periodWeek}. day - ${dayOfWeek}`)

                await getItemsForPeriod(period, dayOfWeek, periodWeek);
            }
        }

        async function getItemsForPeriod(type, day, week) {
            const q = query(collection(db, "days"), where("_periodType", "==", type), where("_dayOfWeek", "==", `${day}`), where("_periodWeekNumber", "==", `${week}`));

            const querySnapshot = await getDocs(q);
            querySnapshot.forEach((doc) => {
                console.log(doc.data())
                days.value.push({
                    id: doc.id,
                    ...doc.data()
                })
            });
        }

        function getMomentDay(val, year) {
            const month = val?.split('.')[1]
            const day = val?.split('.')[0]
            return moment(`${month}.${day}.${year}`, "MM-DD-YYYY");
        }

        function logDay(val) {
            console.log(`selected day: ${val}`)
        }

        function showCalendar() {
            picker.value = selectedDay.value.toDate()
            dialog.value = true
        }

        function loadDay() {
            dialog.value = false
            router.push(`/day/${moment(picker.value).format("DD-MM-YYYY")}`)
        }

        onMounted(fetchItems);
        return {
            days,
            loading,
            selectedDay,
            dayToDisplay,
            dialog,
            picker,
            snackText,
            snackbar,
            dayToURL,

            showCalendar,
            loadDay
        };

    }


}


</script>

// stores/fonts.js
import { defineStore } from 'pinia'

import { useCollection, useFirestore } from 'vuefire'
import { collection, orderBy, query, doc, updateDoc, deleteDoc, setDoc, addDoc } from 'firebase/firestore'

export const useDayStore = defineStore('day', {
    state: () => {
        return {
            dayToEdit: {},
            dayToDelete: {}
        }
    },
    actions: {
        setDayToEdit(dte) {
            this.dayToEdit = dte;
        },
        async updateDay(dte) {
            this.dayToEdit = dte;
            
            const db = useFirestore()
            const docRef = doc(db, "days", `${dte.id}`);

            await updateDoc(docRef, {
                ...dte
            });
        },
        async deleteDay(dte) {
            this.dayToDelete = dte;
            
            const db = useFirestore()
            const docRef = doc(db, "days", `${dte.id}`);

            await deleteDoc(docRef, {
                ...dte
            });
        }
    },

    getters: {
        getDayToEdit: (state) => {
            return state.dayToEdit;
        }
    },
})

      this.headerSize = this.headerSize - 0.05
      this.headerLetterSpacing = this.headerLetterSpacing - 0.005
      this.textLine = this.textLine - 0.03

    },

      console.log(this.textSize)
      this.textLetterSpacing = this.textLetterSpacing + 0.005
      this.textSize = this.textSize + 0.1

      this.textLine = this.textLine + 0.03
