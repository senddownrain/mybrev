function getTableField(table, id, field) {
    let tableName = table + "_table";

    if (tables[tableName]) {

        let item = tables[tableName].find(x => x[table + "_ID"] == id)
        //console.log(item)
        if (item && item[field]) {
            return item[field]
        }
    }
    return ""
}

    async function updateFirestoreDays(uniqueDays) {
        try {
            const querySnapshot = await getDocs(query(daysCollectionRef, orderBy('_weekNum')));
            const batch = [];

            // Проходим по каждому документу из Firestore
            querySnapshot.forEach(docSnapshot => {
                const docData = docSnapshot.data();
                const matchingUniqueDay = uniqueDays.find(uniqueDay => uniqueDay.Day === docData._dayName && uniqueDay.weekDay === docData._weekDay);

                // Если найден соответствующий uniqueDay, обновляем officeOfReadingsVers
                if (matchingUniqueDay) {
                    const docRef = doc(db, 'days', docSnapshot.id);
                    const updateOperation = updateDoc(docRef, {
                        officeOfReadingsVers: matchingUniqueDay.officeOfReadingsVers
                    });
                    // console.log(`${docData._dayName}-${docData._weekDay} updated`)
                    batch.push(updateOperation);
                }
            });

            console.log(`to update ${batch.length}`);
            // Выполняем все обновления
            await Promise.all(batch);

            console.log("Обновление завершено.");
        } catch (error) {

        }
    }

  { title: 'Дзеянні', key: 'actions', sortable: false },

// --- Форма добавления ---
const defaultItem = { year: new Date().getFullYear() + 1, ashWednesday: "", easterSunday: "", ascension: '', pentecost: "", corpusChristi: "", firstAdventSunday: "", ordintaryStart1: "", ordintaryWeeks1: "", ordintaryStart2: '', ordintaryWeeks2: "" };
const newItem = reactive({ ...defaultItem });

const isNewItemValid = computed(() => newItem.year && newItem.easterSunday);

// --- Полный калькулятор ---
const calculateAllDates = () => {
  const y = newItem.year;
  if (!y || y < 1900 || y > 2200) return;
  // --- Вспомогательные функции для работы в UTC ---
  const formatDate = (date) => {
    const year = date.getUTCFullYear();
    const month = (date.getUTCMonth() + 1).toString().padStart(2, '0');
    const day = date.getUTCDate().toString().padStart(2, '0');
    return `${year}-${month}-${day}`;
  };
  const addDays = (d, days) => {
    const date = new Date(d);
    date.setUTCDate(date.getUTCDate() + days);
    return date;
  };
  
  // 1. Вычисляем Пасху (католическую) строго в UTC
  const a = y % 19;
  const b = Math.floor(y / 100);
  const c = y % 100;
  const d = Math.floor(b / 4);
  const e = b % 4;
  const f = Math.floor((b + 8) / 25);
  const g = Math.floor((b - f + 1) / 3);
  const h = (19 * a + b - d - g + 15) % 30;
  const i = Math.floor(c / 4);
  const k = c % 4;
  const l = (32 + 2 * e + 2 * i - h - k) % 7;
  const m = Math.floor((a + 11 * h + 22 * l) / 451);
  const month = Math.floor((h + l - 7 * m + 114) / 31);
  const day = ((h + l - 7 * m + 114) % 31) + 1;
  const easterDate = new Date(Date.UTC(y, month - 1, day));
  
  newItem.easterSunday = formatDate(easterDate);
  
  // 2. Вычисляем все остальные даты относительно Пасхи
  const ashWednesdayDate = addDays(easterDate, -46);
  newItem.ashWednesday = formatDate(ashWednesdayDate);
  newItem.ascension = formatDate(addDays(easterDate, 39));
  newItem.pentecost = formatDate(addDays(easterDate, 49));
  newItem.corpusChristi = formatDate(addDays(easterDate, 60));
  // 3. Вычисляем Адвент
  const christmas = new Date(Date.UTC(y, 11, 25));
  const firstSundayOfAdvent = addDays(christmas, -christmas.getUTCDay() - 21);
  newItem.firstAdventSunday = formatDate(firstSundayOfAdvent);
  // 4. Вычисляем Рядовые периоды
  const epiphany = new Date(Date.UTC(y, 0, 6)); // Богоявление
  let baptismLord = addDays(epiphany, 7 - epiphany.getUTCDay());
  if (baptismLord.getUTCMonth() === 0 && baptismLord.getUTCDate() < 7) {
      baptismLord = addDays(baptismLord, 7); // Если крещение слишком рано, переносим на неделю
  }
  
  const ordinaryStart1Date = addDays(baptismLord, 1); // Начало 1-го периода
  newItem.ordintaryStart1 = formatDate(ordinaryStart1Date);
  const ordinaryEnd1Date = addDays(ashWednesdayDate, -1); // Вторник перед Папяльцом
  
  // Неделя, на которой заканчивается 1-й период.
  // Это номер недели Рядового времени, а не календарной.
  const ordinaryWeeks1 = Math.ceil((ordinaryEnd1Date.getTime() - ordinaryStart1Date.getTime() + (1000 * 60 * 60 * 24)) / (1000 * 60 * 60 * 24 * 7));
  newItem.ordintaryWeeks1 = ordinaryWeeks1;
  // Начало 2-го периода
  const ordinaryStart2Date = addDays(new Date(newItem.pentecost), 1);
  newItem.ordintaryStart2 = formatDate(ordinaryStart2Date);
  
  // Номер недели, с которой возобновляется период
  newItem.ordintaryWeeks2 = ordinaryWeeks1 + 2;
};

// --- CRUD ---

  // Двойная проверка, хотя кнопка и так должна быть неактивна

    // 1. Сбрасываем форму к значениям по умолчанию

    newItem.year = new Date().getFullYear() + 1;

    alert("Не атрымалася дадаць год. Глядзіце кансоль для падрабязнасцяў.");
  }
};
/**
 * Удаляет выбранный год из Firestore

<script setup>
import { ref, watch, computed } from 'vue';

  console.log(`[ItemsList] Запускаем загрузку для даты: ${date.format('YYYY-MM-DD')}`);
  loading.value = true;
  days.value = [];    // Очищаем 'days'
  error.value = null; // Сбрасываем ошибку
  try {
     const result = await getLiturgicalOptionsForDate(date.toDate());
    if (result && result.items) {
      // ✅ ПРИСВАИВАЕМ ИМЕННО МАССИВ 'items'
      days.value = result.items;
    } else {
      // На случай, если что-то пошло не так
      days.value = [];
    }

    if (days.value.length === 0) {
      console.warn(`[ItemsList] Для даты ${date.format('YYYY-MM-DD')} не найдено ни одного литургического элемента.`);
    }
  } catch (e) {
    console.error("Произошла критическая ошибка при загрузке данных:", e);

    loading.value = false;
  }

// --- Навигация ---
function navigateToDate(date) {
  const dateString = date.format('YYYY-MM-DD');
  // Явно указываем имя маршрута 'SelectedDay'
  router.push({ name: 'SelectedDay', params: { day: dateString } });

### src/components/oraparts/DayTitle.vue

<template>
  <!-- Используем наш глобальный класс для автоматического применения стилей -->
  <div class="prayer-content-area text-center mb-6">
    
    <!-- Индикатор загрузки -->
    <v-skeleton-loader v-if="loading" type="heading, subtitle"></v-skeleton-loader>

    <!-- Основной контент, когда данные загружены -->
    <div v-else-if="dayData">
      <!-- 1. Полная дата -->
               <div class="text-subtitle-1 text-grey-darken-1" v-html="formattedDate"></div>
        
      
      <!-- 2. Название праздника или дня периода -->
      <h1 class="text-h4 font-weight-light mt-1">
        {{ dayData._feastName || getPeriodTitle(dayData) }}
      </h1>
      
      <!-- 3. Ранг праздника или псалтырь -->
      <p v-if="dayData._feastType || dayData._psalmWeek" class="text-subtitle-1 mt-1">
        {{ dayData._feastType ? `${getRankName(dayData._feastType)}` : `${dayData._psalmWeek}` }}
      </p>

<div v-if="dayData.feastComment" lass="text-subtitle-1 mt-1" v-html="dayData.feastComment"></div>
     
    </div>

    <!-- Сообщение, если данные не загрузились -->
    <div v-else class="text-grey">
      Немагчыма вызначыць назву дня
    </div>
  </div>
</template>

<script setup>
import { computed } from 'vue';
import { useRoute } from 'vue-router';
import { useDocument } from 'vuefire';
import { doc, getFirestore } from 'firebase/firestore';
import moment from 'moment';
import 'moment/dist/locale/be'; // Убедимся, что локаль подключена

// --- Инициализация ---
const route = useRoute();
const db = getFirestore();
moment.locale('be'); // Активируем белорусскую локаль

// --- Загрузка данных дня ---
// Получаем ID дня из URL
const dayId = route.params.id; 
// Загружаем документ
const dayDocRef = doc(db, 'days', dayId);
const { data: dayData, pending: loading } = useDocument(dayDocRef);

// --- Форматирование даты ---
// Берем полную дату из параметра URL (:day) и форматируем ее
const formattedDate = computed(() => {
  if (!route.params.day) return '';
  // Используем массив форматов для надежности, как мы делали раньше
  const mDate = moment(route.params.day, ['YYYY-MM-DD', 'DD-MM-YYYY']);
  if (!mDate.isValid()) return '';
  // Форматируем в красивый вид, например, "18 верасня 2025 г."
  const dateStr = mDate.format('D MMMM YYYY г.');
  return dateStr.charAt(0).toUpperCase() + dateStr.slice(1);
});

// --- Вспомогательные функции для отображения ---

// Перевод ранга на белорусский
const getRankName = (rank) => {
  const rankMap = {
    solemnity: 'Урачыстасць',
    feast: 'Свята',
    memorial: 'Успамін'
  };
  return rankMap[rank] || rank;
};

// Формирование заголовка для дня периода
const getPeriodTitle = (day) => {
  if (!day || !day._periodType) return 'Літургія Гадзінаў';

  const periodMap = {
    ordinary: 'Звычайны перыяд',
    advent: 'Адвэнт',
    lent: 'Вялікі Пост',
    easter: 'Велікодны перыяд',
    christmas: 'Перыяд Нараджэння Пана',
    holyweek: 'Вялікі тыдзень'
  };
  const periodName = periodMap[day._periodType] || 'Перыяд';
  const dayOfWeekMap = { '1': 'Нядзеля', '2': 'Панядзелак', '3': 'Аўторак', '4': 'Серада', '5': 'Чацвер', '6': 'Пятніца', '7': 'Субота' };
  const dayName = dayOfWeekMap[day._dayOfWeek] || '';

  return `${periodName}, ${day._periodWeekNumber} тыдзень, ${dayName}`;
};
</script>

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import { getAuth, onAuthStateChanged, signInWithEmailAndPassword, signOut } from 'firebase/auth';
import router from '@/router'; // ✅ Возвращаем импорт роутера

const ADMIN_UIDS = ['64MeSP56IWVotVFG7bu8YKeLis12']; // ❗️ Не забудьте вставить ваш реальный UID

export const useAuthStore = defineStore('auth', () => {
  const user = ref(null);
  const loading = ref(true);

  const isAdmin = computed(() => user.value && ADMIN_UIDS.includes(user.value.uid));

  // Слушатель onAuthStateChanged остается без изменений
  const init = () => {
    const auth = getAuth();
    onAuthStateChanged(auth, (firebaseUser) => {
      user.value = firebaseUser;
      loading.value = false;
    });
  };

  /**
   * Login с использованием setTimeout
   */
  const login = async (email, password) => {
    const auth = getAuth();
    try {
      await signInWithEmailAndPassword(auth, email, password);
      // Успешный вход. Теперь используем трюк с setTimeout
      setTimeout(() => {
        router.push('/');
      }, 100); // 100мс - более чем достаточно для обновления Vue
    } catch (error) {
      console.error("Памылка ўваходу:", error);
      alert(`Памылка ўваходу: ${error.code}`);
      throw error; // Пробрасываем ошибку, чтобы компонент мог ее поймать
    }
  };

  const logout = async () => {
    const auth = getAuth();
    await signOut(auth);
    // При выходе всегда отправляем на главную
    router.push('/');
  };

  return { user, loading, isAdmin, init, login, logout };
});

### src/stores/calendar.js
```
import { defineStore } from 'pinia';
import { ref, computed  } from 'vue'; // Убедимся, что импорт правильный
import { collection, query, where, getDocs, getFirestore } from 'firebase/firestore';
import moment from 'moment';
import 'moment/dist/locale/be';

export const useCalendarStore = defineStore('calendar', () => {
    const db = getFirestore();
    moment.locale('be');

    const yearOptionsCache = ref({});
    const dynamicDatesCache = ref({});

    // --- ✅ 1. ИЗМЕНЕНИЕ: getYearOptions теперь ищет по полю, а не по ID ---
    async function getYearOptions(year) {
        if (yearOptionsCache.value[year]) {
            return yearOptionsCache.value[year];
        }
        
        console.log(`[CalendarStore] Пошук наладаў для ${year} года ў Firestore...`);
        const q = query(collection(db, 'yearOptions'), where('year', '==', Number(year)));
        
        const querySnapshot = await getDocs(q);
        
        if (querySnapshot.empty) {
            console.error(`[CalendarStore] Налады для года ${year} не знойдзены!`);
            return null;
        }
        
        const options = querySnapshot.docs[0].data();
        yearOptionsCache.value[year] = options;
        console.log(`[CalendarStore] Налады для года ${year} паспяхова загружаны.`);
        return options;
    }

    // --- ✅ 2. ИЗМЕНЕНИЕ: getDynamicDates работает с новым форматом дат ---
    async function getDynamicDates(year) {
    console.log(`[CalendarStore] Вычисление динамических дат для ${year} года...`);
    
    // Убедимся, что опции для года загружены
    await getYearOptions(year); 
    const opts = yearOptionsCache.value[year];
    if (!opts) {
        console.error(`[CalendarStore] Не удалось получить опции для ${year} года. Вычисление дат прервано.`);
        return {};
    }
    const dates = {};
    const getMoment = (isoDateStr) => moment(isoDateStr, 'YYYY-MM-DD');
    const easter = getMoment(opts.easterSunday);
    const pentecost = getMoment(opts.pentecost);
    const firstAdventSunday = getMoment(opts.firstAdventSunday);
    // Расчет дат до Пасхи
    dates.ashWednesday = getMoment(opts.ashWednesday);
    dates.palmSunday = easter.clone().subtract(7, 'days'); // ✅ ВЕРБНОЕ ВОСКРЕСЕНЬЕ
    // Расчет дат после Пасхи
    dates.easterSunday = easter;
    dates.pentecost = pentecost;
    dates.ascension = pentecost.clone().subtract(7, 'days'); // ✅ ВОЗНЕСЕНИЕ (7-е воскресенье Пасхи)
    
    const holyTrinitySunday = pentecost.clone().add(7, 'days');
    dates.holyTrinity = holyTrinitySunday;
    dates.corpusChristi = holyTrinitySunday.clone().add(4, 'days'); // ✅ ТЕЛО И КРОВЬ ХРИСТА (четверг после Троицы)
    // Расчет дат в "обычное время"
    dates.holyFamily = getMoment(opts.holyFamily);
    dates.baptismOfLord = getMoment(opts.baptismOfLord);
    dates.christKing = firstAdventSunday.clone().subtract(7, 'days');
    const sacredHeartOfJesus = pentecost.clone().add(19, 'days');
    dates.sacredHeartOfJesus = sacredHeartOfJesus;
    dates.immaculateHeartOfMary = sacredHeartOfJesus.clone().add(1, 'day'); // Следующий день
    dates.christTheHighPriest = pentecost.clone().subtract(11, 'days'); // Пн после Пятидесятницы - уточнить если нужно
    dynamicDatesCache.value[year] = dates;
    console.log(`[CalendarStore] Динамические даты для ${year} года вычислены.`);
    return dates;
}

    return { 
        getYearOptions, 
        getDynamicDates, 
        // Геттеры для доступа из компонентов
        optionsForYear: (year) => computed(() => yearOptionsCache.value[year]),
        dynamicDatesForYear: (year) => computed(() => dynamicDatesCache.value[year])
    };
});

### src/stores/settings.js
```
import { defineStore } from 'pinia';
import { ref, watch } from 'vue';
import { useTheme } from 'vuetify';

// Константа для системного шрифта
const systemFont = "-apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, Helvetica, Arial, sans-serif";

// Экспортируем константу с правильным именем 'useSettingsStore'
export const useSettingsStore = defineStore('settings', () => {
  const theme = useTheme();

  // --- Состояния (State) ---
  const currentTheme = ref(localStorage.getItem('theme') || 'light');
  const fontFamily = ref(localStorage.getItem('fontFamily') || systemFont);
  const fontSizeMultiplier = ref(parseFloat(localStorage.getItem('fontSizeMultiplier')) || 1.0);

  // --- Слежение за изменениями (Watchers) ---
  watch(currentTheme, (newTheme) => {
    theme.global.name.value = newTheme;
    localStorage.setItem('theme', newTheme);
  }, { immediate: true });

  watch(fontFamily, (newFont) => {
    document.documentElement.style.setProperty('--app-font-family', newFont);
    localStorage.setItem('fontFamily', newFont);
  }, { immediate: true });

  watch(fontSizeMultiplier, (newMultiplier) => {
    document.documentElement.style.setProperty('--font-size-multiplier', newMultiplier);
    localStorage.setItem('fontSizeMultiplier', String(newMultiplier));
  }, { immediate: true });


  // --- Действия (Actions) ---
  function toggleTheme() {
    currentTheme.value = currentTheme.value === 'light' ? 'dark' : 'light';
  }

  function setFontFamily(font) {
    fontFamily.value = font;
  }

  function increaseFontSize() {
    if (fontSizeMultiplier.value < 1.8) {
      fontSizeMultiplier.value = parseFloat((fontSizeMultiplier.value + 0.1).toFixed(2));
    }
  }
  function decreaseFontSize() {
    if (fontSizeMultiplier.value > 0.7) {
      fontSizeMultiplier.value = parseFloat((fontSizeMultiplier.value - 0.1).toFixed(2));
    }
  }

  return {
    currentTheme,
    fontFamily,
    systemFont,
    fontSizeMultiplier,
    toggleTheme,
    setFontFamily,
    increaseFontSize,
    decreaseFontSize,
  };
});

<template>
  <v-container>
    <v-row justify="center">
      <v-col cols="12" sm="8" md="6">
        <v-card class="pa-4">
          <v-card-title class="text-h5 text-center">Уваход для адміністратара</v-card-title>
          <v-card-text>
            <v-form @submit.prevent="handleLogin">
              <v-text-field
                v-model="email"
                label="Email"
                type="email"
                prepend-inner-icon="mdi-email-outline"
                variant="outlined"
                class="mb-2"
              ></v-text-field>
              
              <v-text-field
                v-model="password"
                label="Пароль"
                type="password"
                prepend-inner-icon="mdi-lock-outline"
                variant="outlined"
              ></v-text-field>

            <!-- Внутри <v-form> -->
<v-alert v-if="error" type="error" density="compact" class="my-3" text>{{ error }}</v-alert>
<v-btn 
  type="submit" 
  color="primary" 
  block 
  size="large" 
  class="mt-4"
  :loading="isLoggingIn"
>
  Увайсці
</v-btn>
            </v-form>
          </v-card-text>
        </v-card>
      </v-col>
    </v-row>
  </v-container>
</template>

<script setup>
import { ref } from 'vue';
import { useAuthStore } from '@/stores/auth.js';

const email = ref('');
const password = ref('');
const error = ref(null); // Для отображения ошибки в шаблоне
const isLoggingIn = ref(false); // Для состояния загрузки кнопки

const authStore = useAuthStore();

const handleLogin = async () => {
  if (!email.value || !password.value) {
    error.value = 'Калі ласка, увядзіце email і пароль.';
    return;
  }
  
  isLoggingIn.value = true;
  error.value = null;

  try {
    // Просто вызываем login. Store сам позаботится о редиректе.
    await authStore.login(email.value, password.value);
  } catch (e) {
    // Если store пробросил ошибку, показываем ее
    error.value = 'Няправільны email або пароль.';
  } finally {
    isLoggingIn.value = false;
  }
};
</script>
